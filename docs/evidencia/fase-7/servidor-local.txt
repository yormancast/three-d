import http from 'node:http';
import fs from 'node:fs';
import path from 'node:path';
const root='/Users/yorman/Projects/Repos/three-d/dist';
const out='/Users/yorman/Projects/Repos/three-d/docs/evidencia/fase-7/mediciones-locales.jsonl';
let warm=false;
const types={'.html':'text/html','.js':'text/javascript','.css':'text/css','.glb':'model/gltf-binary','.jpg':'image/jpeg','.ico':'image/x-icon'};
http.createServer(async(req,res)=>{
  const url=new URL(req.url,'http://127.0.0.1:4180');
  if(url.pathname==='/__qa/results' && req.method==='POST'){
    let body='';for await(const chunk of req){body+=chunk;if(body.length>1000000){res.writeHead(413).end();return;}}
    try{const value=JSON.parse(body);fs.appendFileSync(out,JSON.stringify(value)+'\n');res.writeHead(204).end();}catch{res.writeHead(400).end();}return;
  }
  if(url.pathname==='/__qa/instrument.js'){res.writeHead(200,{'Content-Type':'text/javascript','Cache-Control':'no-store'}).end(fs.readFileSync('/tmp/three-d-qa/instrument.js'));return;}
  if(url.pathname==='/')warm=url.searchParams.get('cache')==='warm';
  const file=path.resolve(root,'.'+decodeURIComponent(url.pathname==='/'?'/index.html':url.pathname));
  if(!file.startsWith(root+path.sep)){res.writeHead(403).end();return;}
  try{
    let data=fs.readFileSync(file);
    if(file.endsWith('index.html'))data=Buffer.from(data.toString().replace('<head>','<head><script src="/__qa/instrument.js"></script>'));
    res.writeHead(200,{'Content-Type':types[path.extname(file)]||'application/octet-stream','Cache-Control':warm&&!file.endsWith('index.html')?'public,max-age=3600':'no-store'}).end(data);
  }catch{res.writeHead(404).end('Not found');}
}).listen(4180,'127.0.0.1',()=>console.log('QA local: http://127.0.0.1:4180/'));
